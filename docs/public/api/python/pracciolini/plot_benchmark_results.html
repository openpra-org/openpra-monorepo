119def load_timing(results_dir, pattern, param_key, ext):
120    path = _find_file(results_dir, pattern)
121    if not path:
122        return {}
123    try:
124        with open(path, encoding="utf-8") as f:
125            data = json.load(f)
126    except Exception:
127        return {}
128    out = {}
129    for entry in data.get("results", []):
130        params = entry.get("parameters", {})
131        raw = params.get(param_key) or next(iter(params.values()), "")
132        if not raw:
133            continue
134        model = _model_name(raw, ext)
135        out[model] = {
136            "mean":      entry.get("mean", TIMEOUT_S),
137            "timed_out": 124 in entry.get("exit_codes", []),
138        }
139    return out

236def load_values(results_dir, models):
237    values_all = {}
238    for label, out_dir_name, _, *_ in OUTPUT_SOURCES:
239        out_dir = os.path.join(results_dir, out_dir_name)
240        reader_fn = _READERS.get(label)
241        if not reader_fn:
242            continue
243        per_model = {}
244        for model in models:
245            prob, mcs = reader_fn(out_dir, model)
246            if prob is not None or mcs is not None:
247                per_model[model] = (prob, mcs)
248        values_all[label] = per_model
249    return values_all

252def load_model_stats(data_dir, model_name):
253    if not data_dir:
254        return None
255    path = os.path.join(data_dir, f"{model_name}.xml")
256    if not os.path.exists(path):
257        return None
258    try:
259        from lxml import etree
260        tree = etree.parse(path)
261    except Exception:
262        return None
263    gate_type_counts = {}
264    total_gates = 0
265    for elem in tree.iter("define-gate"):
266        total_gates += 1
267        children = list(elem)
268        if children:
269            tag = children[0].tag
270            display = _GATE_DISPLAY.get(tag, tag.upper())
271            gate_type_counts[display] = gate_type_counts.get(display, 0) + 1
272    total_bes = sum(1 for _ in tree.iter("define-basic-event"))
273    return {"total_gates": total_gates, "gate_types": gate_type_counts, "total_bes": total_bes}

316def _fig_spec(model, timing_all, values_all):
317    labels, times, colors, texts = [], [], [], []
318    group_boundaries = []
319
320    for group_name, subgroups in TIMING_GROUPS:
321        group_start = len(labels)
322        for subgroup_labels in subgroups:
323            available = [
324                (lbl, t)
325                for lbl in subgroup_labels
326                for t in [timing_all.get(lbl, {}).get(model)]
327                if t is not None
328                and not t["timed_out"]
329                and values_all.get(lbl, {}).get(model) is not None
330            ]
331            available.sort(key=lambda x: x[1]["mean"])
332            for label, t in available:
333                mean_ms   = t["mean"] * 1000
334                timed_out = t["timed_out"]
335                labels.append(label)
336                times.append(min(mean_ms, TIMEOUT_MS))
337                colors.append(_rgba(TIMEOUT_COLOR, 0.85) if timed_out else _rgba(_solver_color(label)))
338                texts.append("TIMEOUT" if timed_out else f"{mean_ms:.1f} ms")
339        group_end = len(labels) - 1
340        if group_start <= group_end:
341            group_boundaries.append((group_start, group_end, group_name))
342
343    if not labels:
344        return None
345
346    fig = go.Figure(go.Bar(
347        x=labels, y=times,
348        marker=dict(color=colors, line=dict(width=0)),
349        text=texts, textposition="outside", textfont=dict(size=9),
350        hovertemplate="<b>%{x}</b><br>%{y:.2f} ms<extra></extra>",
351    ))
352
353    fig.add_shape(
354        type="line",
355        x0=-0.5, x1=len(labels) - 0.5,
356        y0=TIMEOUT_MS, y1=TIMEOUT_MS,
357        line=dict(color=TIMEOUT_COLOR, width=1.5, dash="dash"),
358        layer="above",
359    )
360    fig.add_annotation(
361        x=len(labels) - 1, y=TIMEOUT_MS,
362        text=f"Timeout ({int(TIMEOUT_S)}s)",
363        showarrow=False, yanchor="bottom", xanchor="right",
364        font=dict(size=9, color=TIMEOUT_COLOR),
365        bgcolor="rgba(255,255,255,0.75)",
366    )
367
368    for i, (start, end, name) in enumerate(group_boundaries):
369        if i > 0:
370            sep_x = start - 0.5
371            fig.add_shape(
372                type="line",
373                x0=sep_x, x1=sep_x, y0=0, y1=1,
374                xref="x", yref="paper",
375                line=dict(color="#c5cff9", width=1.5, dash="dot"),
376            )
377        mid = (start + end) / 2
378        fig.add_annotation(
379            x=mid, y=1.04, xref="x", yref="paper",
380            text=name, showarrow=False,
381            xanchor="center", yanchor="bottom",
382            font=dict(size=8, color="#495057", family=_FONT["family"]),
383            bgcolor="rgba(240,242,245,0.85)",
384            bordercolor="#dee2e6", borderwidth=1, borderpad=2,
385        )
386
387    fig.update_layout(
388        template="plotly_white", height=460,
389        yaxis=dict(title="Execution time (ms)", gridcolor="#dee2e6", type="log"),
390        xaxis=dict(tickangle=-38, tickfont=dict(size=9)),
391        margin=dict(l=70, r=10, t=50, b=130),
392        font=_FONT, showlegend=False,
393        plot_bgcolor="white", paper_bgcolor="white",
394    )
395    return fig.to_json().replace("</", "<\\/")

398def _combined_table(model, values_all, timing_all):
399    rows = ""
400    for group_name, subgroups in OUTPUT_GROUPS:
401        group_rows = ""
402        for subgroup_labels in subgroups:
403            available = [l for l in subgroup_labels if values_all.get(l, {}).get(model) is not None]
404            available.sort(key=lambda l: _get_timing_mean(l, model, timing_all))
405            for label in available:
406                prob_val, mcs_val = values_all[label][model]
407                if prob_val is None and mcs_val is None:
408                    continue
409                is_prob_ref, is_mcs_ref = _REF_FLAGS.get(label, (False, False))
410                color   = _solver_color(label)
411                is_ref  = is_prob_ref or is_mcs_ref
412                ref_cls = ' class="ref-row"' if is_ref else ""
413                note    = (" †" if is_prob_ref else "") + (" ‡" if is_mcs_ref else "")
414                prob_cell = (
415                    f'<td style="font-family:monospace;text-align:right">{prob_val:.6e}</td>'
416                    if prob_val is not None else
417                    '<td style="text-align:right;color:#aaa">—</td>'
418                )
419                mcs_cell = (
420                    f'<td style="font-family:monospace;text-align:right">{int(mcs_val):,}</td>'
421                    if mcs_val is not None else
422                    '<td style="text-align:right;color:#aaa">—</td>'
423                )
424                group_rows += (
425                    f'<tr{ref_cls}>'
426                    f'<td><span class="solver-dot" style="background:{color}"></span>{label}{note}</td>'
427                    + prob_cell + mcs_cell +
428                    "</tr>"
429                )
430        if group_rows:
431            rows += f'<tr class="group-header"><td colspan="3">{group_name}</td></tr>'
432            rows += group_rows
433
434    if not rows:
435        return '<p class="note">No data available</p>'
436    return (
437        "<table>"
438        "<thead><tr>"
439        "<th>Tool — Algorithm</th>"
440        '<th style="text-align:right">Top Event Probability</th>'
441        '<th style="text-align:right">Minimal Cut Sets</th>'
442        "</tr></thead>"
443        f"<tbody>{rows}</tbody>"
444        "</table>"
445        '<p class="note">† Probability reference (SCRAM BDD) &nbsp; ‡ MCS reference (SCRAM ZBDD REA)</p>'
446    )

449def _timeout_banner(model, timing_all):

460        '<div class="timeout-banner">'
461        f'<strong>Timed out ({n} algorithm{"s" if n > 1 else ""})</strong>'
462        f" &mdash; exceeded {int(TIMEOUT_S)}s limit: {tags}"
463        "</div>"
